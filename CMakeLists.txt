cmake_minimum_required(VERSION 3.20)
project(m1pn_element_weno LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# M1PN solver: element WENO on tensor Bernstein elements (docs/ELEMENT_WENO.md). No MFEM since 2026-11-16 (the
# mass-correction/biharmonic paths were removed; last state with them: tag standard-20261116). Needs Eigen, OpenMP and
# optionally MPI (any OpenMPI, e.g. the UCX/InfiniBand one of the spack environment m1pn-ucx) and CUDA (--gpu).
option(ENABLE_MPI "MPI slabs (docs/ELEMENT_WENO_MPI.md)" ON)
option(M1PN_ELEMENT_GPU "Element WENO GPU time loop (CUDA, docs/ELEMENT_WENO_GPU.md)" OFF)
# AVX2/FMA (x86-64-v3, NX and tardis). Contraction only shrinks each operation's rounding error, so the exact-cone filter
# bound in src/cone.hpp stays valid. Same flags as the former m1pn_core: results bitwise equal to standard-20261116.
option(M1PN_OPTIMIZED_ISA "Target AVX2/FMA (x86-64-v3)" ON)
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
find_package(OpenMP REQUIRED)
add_library(m1pn_element src/angular.cpp src/boundary.cpp src/element_weno.cpp src/positivity_projection.cpp
            src/separable_trace.cpp src/element_weno_mpi.cpp)
target_include_directories(m1pn_element PUBLIC src)
target_link_libraries(m1pn_element PUBLIC Eigen3::Eigen OpenMP::OpenMP_CXX)
set(M1PN_OPT_FLAGS -fno-fast-math -fno-math-errno -fno-trapping-math)
if(M1PN_OPTIMIZED_ISA AND CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64")
  list(APPEND M1PN_OPT_FLAGS -march=x86-64-v3 -mtune=znver3 -ffp-contract=fast)
else()
  list(APPEND M1PN_OPT_FLAGS -ffp-contract=off)
endif()
target_compile_options(m1pn_element PUBLIC ${M1PN_OPT_FLAGS})
target_compile_options(m1pn_element PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(m1pn_element PUBLIC EIGEN_DONT_PARALLELIZE)
if(ENABLE_MPI)
  find_package(MPI REQUIRED COMPONENTS CXX)
  target_link_libraries(m1pn_element PUBLIC MPI::MPI_CXX)
  target_compile_definitions(m1pn_element PUBLIC M1PN_MPI)
endif()
if(M1PN_ELEMENT_GPU)
  if(NOT CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 80) # A100 (tardis)
  endif()
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_library(m1pn_element_gpu STATIC src/element_weno_gpu.cu)
  set_target_properties(m1pn_element_gpu PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_compile_options(m1pn_element_gpu PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3 -lineinfo>)
  target_link_libraries(m1pn_element_gpu PUBLIC CUDA::cudart CUDA::cublas)
  target_sources(m1pn_element PRIVATE src/element_weno_gpu.cpp)
  target_link_libraries(m1pn_element PUBLIC m1pn_element_gpu)
  target_compile_definitions(m1pn_element PUBLIC M1PN_ELEMENT_GPU)
endif()

# Programs: m1pn_fast (standard: all cases, MPI, --gpu) and m1pn_element_weno (stand-alone studies: smooth/periodic
# accuracy, smooth outflow, ghost study, step cache).
add_executable(m1pn_fast src/fast_main.cpp)
target_link_libraries(m1pn_fast PRIVATE m1pn_element)
add_executable(m1pn_element_weno src/element_weno_main.cpp)
target_link_libraries(m1pn_element_weno PRIVATE m1pn_element)
add_executable(m1pn_element_weno_time_study tests/element_weno_time_study.cpp)
target_link_libraries(m1pn_element_weno_time_study PRIVATE m1pn_element)
add_executable(m1pn_positivity_projection_bench tests/positivity_projection_bench.cpp)
target_link_libraries(m1pn_positivity_projection_bench PRIVATE m1pn_element)
# Monte-Carlo reference for lattice-paul and lattice-3d (docs/LATTICE_MC.md); independent of the solver, it only shares
# the material predicates (src/benchmark_cases.hpp) and, for its validation, the uncollided analytics.
add_executable(m1pn_lattice_mc tools/lattice_mc.cpp)
target_link_libraries(m1pn_lattice_mc PRIVATE m1pn_element)
# Study of the WENO activation at the rounding limit (smooth periodic case; docs/ELEMENT_WENO.md). Not a ctest.
add_executable(m1pn_weno_roundoff_study tests/weno_roundoff_study.cpp)
target_link_libraries(m1pn_weno_roundoff_study PRIVATE m1pn_element)

include(CTest)
if(BUILD_TESTING)
  function(m1pn_test name source threads timeout)
    add_executable(${name}_tests ${source})
    target_link_libraries(${name}_tests PRIVATE m1pn_element)
    add_test(NAME ${name} COMMAND ${name}_tests)
    set_tests_properties(${name} PROPERTIES ENVIRONMENT "OMP_NUM_THREADS=${threads};OPENBLAS_NUM_THREADS=1"
                         TIMEOUT ${timeout})
  endfunction()
  m1pn_test(element_weno tests/element_weno.cpp 1 300)
  m1pn_test(element_weno_material tests/element_weno_material.cpp 2 600)
  m1pn_test(element_weno_fixes tests/element_weno_fixes.cpp 1 300)
  m1pn_test(element_weno_fcs tests/element_weno_fcs.cpp 1 300)
  m1pn_test(uncollided_lattice tests/uncollided_lattice.cpp 2 300)
  m1pn_test(uncollided_lattice3d tests/uncollided_lattice3d.cpp 2 300)
  m1pn_test(positivity_projection tests/positivity_projection.cpp 1 300)
  if(ENABLE_MPI)
    # Run with mpirun (not in ctest): mpirun -np 4 m1pn_element_weno_mpi_tests
    add_executable(m1pn_element_weno_mpi_tests tests/element_weno_mpi.cpp)
    target_link_libraries(m1pn_element_weno_mpi_tests PRIVATE m1pn_element)
  endif()
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME fast_cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/fast_cli.py
             $<TARGET_FILE:m1pn_fast> $<TARGET_FILE:m1pn_element_weno>)
    set_tests_properties(fast_cli PROPERTIES TIMEOUT 600)
    add_test(NAME lattice_mc COMMAND ${Python3_EXECUTABLE} -I ${CMAKE_CURRENT_SOURCE_DIR}/tests/lattice_mc_check.py
             $<TARGET_FILE:m1pn_lattice_mc>)
    set_tests_properties(lattice_mc PROPERTIES TIMEOUT 300)
    add_test(NAME campaign_scripts COMMAND ${Python3_EXECUTABLE} -I ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_nx_convergence.py
             WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/scripts/element_weno)
  endif()
endif()
