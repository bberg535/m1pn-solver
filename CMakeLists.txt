cmake_minimum_required(VERSION 3.20)
project(m1pn_lazy_einstrom LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(ENABLE_OPENMP "Parallel angular solves and pair evaluation" ON)
option(ENABLE_MPI "MPI execution (also required by MPI-enabled MFEM)" ON)
set(MFEM_CONFIG "" CACHE FILEPATH "Existing MFEM share/mfem/config.mk; no dependency build")
if(NOT EXISTS "${MFEM_CONFIG}")
  message(FATAL_ERROR "Set MFEM_CONFIG to an existing compatible MFEM config.mk")
endif()
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
execute_process(COMMAND make --no-print-directory -s -f ${CMAKE_CURRENT_SOURCE_DIR}/scripts/mfem-flags.mk
  MFEM_CONFIG=${MFEM_CONFIG} flags OUTPUT_VARIABLE mfem_flags RESULT_VARIABLE mfem_result)
if(NOT mfem_result EQUAL 0)
  message(FATAL_ERROR "Could not read installed MFEM configuration")
endif()
string(REPLACE "\n" ";" mfem_lines "${mfem_flags}")
list(GET mfem_lines 0 mfem_includes)
list(GET mfem_lines 1 mfem_libraries)
list(GET mfem_lines 2 mfem_mpi)
separate_arguments(mfem_includes UNIX_COMMAND "${mfem_includes}")
list(TRANSFORM mfem_includes REPLACE "^-I" "-isystem")
separate_arguments(mfem_libraries UNIX_COMMAND "${mfem_libraries}")
if(mfem_mpi STREQUAL "YES" AND NOT ENABLE_MPI)
  message(FATAL_ERROR "This MFEM is MPI-enabled; configure ENABLE_MPI=ON")
endif()
add_library(m1pn_core src/angular.cpp src/boundary.cpp src/spatial.cpp src/ghost_local.cpp src/stabilization.cpp src/transport.cpp src/time.cpp src/imex.cpp)
target_include_directories(m1pn_core PUBLIC src)
target_link_libraries(m1pn_core PUBLIC Eigen3::Eigen ${mfem_libraries})
target_compile_options(m1pn_core PUBLIC ${mfem_includes} -ffp-contract=off -fno-fast-math)
target_compile_options(m1pn_core PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(m1pn_core PUBLIC EIGEN_DONT_PARALLELIZE)
if(ENABLE_MPI)
  find_package(MPI REQUIRED COMPONENTS CXX)
  target_link_libraries(m1pn_core PUBLIC MPI::MPI_CXX)
  target_compile_definitions(m1pn_core PUBLIC M1PN_MPI)
endif()
if(ENABLE_OPENMP)
  find_package(OpenMP REQUIRED)
  target_link_libraries(m1pn_core PUBLIC OpenMP::OpenMP_CXX)
endif()
add_executable(m1pn src/main.cpp)
target_link_libraries(m1pn PRIVATE m1pn_core)
target_compile_definitions(m1pn PRIVATE PN_AFC_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(m1pn_benchmark tests/benchmark.cpp)
target_link_libraries(m1pn_benchmark PRIVATE m1pn_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(m1pn_ghost_tests tests/ghost_local.cpp)
  target_link_libraries(m1pn_ghost_tests PRIVATE m1pn_core)
  add_test(NAME ghost_local COMMAND m1pn_ghost_tests)
  add_executable(m1pn_ghost_coupling_tests tests/ghost_coupling.cpp)
  target_link_libraries(m1pn_ghost_coupling_tests PRIVATE m1pn_core)
  add_test(NAME ghost_coupling COMMAND m1pn_ghost_coupling_tests)
  add_executable(m1pn_ghost_diagnostics tests/ghost_diagnostics.cpp)
  target_link_libraries(m1pn_ghost_diagnostics PRIVATE m1pn_core)
  add_test(NAME ghost_diagnostics COMMAND m1pn_ghost_diagnostics)
  set_tests_properties(ghost_diagnostics PROPERTIES ENVIRONMENT "OMP_NUM_THREADS=1;OPENBLAS_NUM_THREADS=1" TIMEOUT 180)
  set_tests_properties(ghost_local ghost_coupling PROPERTIES ENVIRONMENT "OMP_NUM_THREADS=2;OPENBLAS_NUM_THREADS=1" TIMEOUT 180)
  add_executable(m1pn_stabilization_tests tests/stabilization.cpp)
  target_link_libraries(m1pn_stabilization_tests PRIVATE m1pn_core)
  add_test(NAME r2_stabilization COMMAND m1pn_stabilization_tests)
  add_executable(m1pn_r2_tests tests/r2_transport.cpp)
  target_link_libraries(m1pn_r2_tests PRIVATE m1pn_core)
  add_test(NAME r2_transport COMMAND m1pn_r2_tests)
  add_executable(m1pn_tests tests/checks.cpp)
  target_link_libraries(m1pn_tests PRIVATE m1pn_core)
  foreach(suite angular boundary spatial coupling time cone_filter base_bar pair_limiter)
    add_test(NAME ${suite} COMMAND m1pn_tests ${suite})
    set_tests_properties(${suite} PROPERTIES ENVIRONMENT "OMP_NUM_THREADS=2;OPENBLAS_NUM_THREADS=1" TIMEOUT 180)
  endforeach()
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME ghost_cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/ghost_cli.py $<TARGET_FILE:m1pn>)
    set_tests_properties(ghost_cli PROPERTIES ENVIRONMENT "OMP_NUM_THREADS=1;OPENBLAS_NUM_THREADS=1" TIMEOUT 60)
    add_test(NAME cli_guards COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_guards.py $<TARGET_FILE:m1pn>)
    set_tests_properties(cli_guards PROPERTIES ENVIRONMENT "OMP_NUM_THREADS=1;OPENBLAS_NUM_THREADS=1" TIMEOUT 60)
    add_test(NAME crossing_2d COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/crossing_2d.py $<TARGET_FILE:m1pn>)
    set_tests_properties(crossing_2d PROPERTIES ENVIRONMENT "OMP_NUM_THREADS=1;OPENBLAS_NUM_THREADS=1" TIMEOUT 60)
    add_test(NAME crossing_three COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/crossing_three.py $<TARGET_FILE:m1pn>)
    set_tests_properties(crossing_three PROPERTIES ENVIRONMENT "OMP_NUM_THREADS=1;OPENBLAS_NUM_THREADS=1" TIMEOUT 180)
  endif()
endif()
