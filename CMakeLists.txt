cmake_minimum_required(VERSION 3.20)
project(m1pn_lazy_einstrom LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(M1PN_ENABLE_CUDA "Compile the narrow sm_86 CUDA path for angular products, scatter, and lumped mass correction" OFF)
if(M1PN_ENABLE_CUDA)
  set(CMAKE_CUDA_COMPILER /usr/local/cuda/bin/nvcc CACHE FILEPATH "nvcc")
  set(CMAKE_CUDA_ARCHITECTURES 86 CACHE STRING "CUDA architectures" FORCE)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
endif()
option(ENABLE_OPENMP "Parallel angular solves and pair evaluation" ON)
option(ENABLE_MPI "MPI execution (also required by MPI-enabled MFEM)" ON)
set(MFEM_CONFIG "" CACHE FILEPATH "Existing MFEM share/mfem/config.mk; no dependency build")
if(NOT EXISTS "${MFEM_CONFIG}")
  message(FATAL_ERROR "Set MFEM_CONFIG to an existing compatible MFEM config.mk")
endif()
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
execute_process(COMMAND make --no-print-directory -s -f ${CMAKE_CURRENT_SOURCE_DIR}/scripts/mfem-flags.mk
  MFEM_CONFIG=${MFEM_CONFIG} flags OUTPUT_VARIABLE mfem_flags RESULT_VARIABLE mfem_result)
if(NOT mfem_result EQUAL 0)
  message(FATAL_ERROR "Could not read installed MFEM configuration")
endif()
string(REPLACE "\n" ";" mfem_lines "${mfem_flags}")
list(GET mfem_lines 0 mfem_includes)
list(GET mfem_lines 1 mfem_libraries)
list(GET mfem_lines 2 mfem_mpi)
separate_arguments(mfem_includes UNIX_COMMAND "${mfem_includes}")
list(TRANSFORM mfem_includes REPLACE "^-I" "-isystem")
separate_arguments(mfem_libraries UNIX_COMMAND "${mfem_libraries}")
if(mfem_mpi STREQUAL "YES" AND NOT ENABLE_MPI)
  message(FATAL_ERROR "This MFEM is MPI-enabled; configure ENABLE_MPI=ON")
endif()
add_library(m1pn_core src/angular.cpp src/boundary.cpp src/spatial.cpp src/ghost_local.cpp src/stabilization.cpp src/transport.cpp src/time.cpp src/imex.cpp src/cuda_path.cpp src/fast_mc.cpp src/element_weno.cpp src/element_weno_dlra.cpp src/positivity_projection.cpp src/separable_trace.cpp)
if(M1PN_ENABLE_CUDA)
  add_library(m1pn_cuda STATIC src/cuda_kernels.cu)
  set_target_properties(m1pn_cuda PROPERTIES CUDA_ARCHITECTURES 86 CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_include_directories(m1pn_cuda PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
  target_compile_options(m1pn_cuda PRIVATE --fmad=false -Xcompiler=-fno-fast-math -Xcompiler=-ffp-contract=off)
  target_link_libraries(m1pn_cuda PUBLIC CUDA::cudart)
  target_link_libraries(m1pn_core PUBLIC m1pn_cuda)
  target_compile_definitions(m1pn_core PUBLIC M1PN_HAS_CUDA=1)
endif()
target_include_directories(m1pn_core PUBLIC src)
target_link_libraries(m1pn_core PUBLIC Eigen3::Eigen ${mfem_libraries})
target_compile_options(m1pn_core PUBLIC ${mfem_includes})
# Scoped to the optimized solver. The frozen baseline/reference cores keep their
# strict -ffp-contract=off build. Contraction only shrinks each operation's
# rounding error, so the exact-cone filter bound in cone.hpp stays valid.
option(M1PN_OPTIMIZED_ISA "Target AVX2/FMA (x86-64-v3, available on the NX nodes)" ON)
set(M1PN_OPT_FLAGS -fno-fast-math -fno-math-errno -fno-trapping-math)
if(M1PN_OPTIMIZED_ISA AND CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64")
  list(APPEND M1PN_OPT_FLAGS -march=x86-64-v3 -mtune=znver3 -ffp-contract=fast)
else()
  list(APPEND M1PN_OPT_FLAGS -ffp-contract=off)
endif()
target_compile_options(m1pn_core PUBLIC ${M1PN_OPT_FLAGS})
target_compile_options(m1pn_core PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(m1pn_core PUBLIC EIGEN_DONT_PARALLELIZE)
if(ENABLE_MPI)
  find_package(MPI REQUIRED COMPONENTS CXX)
  target_link_libraries(m1pn_core PUBLIC MPI::MPI_CXX)
  target_compile_definitions(m1pn_core PUBLIC M1PN_MPI)
endif()
if(ENABLE_OPENMP)
  find_package(OpenMP REQUIRED)
  target_link_libraries(m1pn_core PUBLIC OpenMP::OpenMP_CXX)
endif()
add_executable(m1pn src/main.cpp)
target_link_libraries(m1pn PRIVATE m1pn_core)
target_compile_definitions(m1pn PRIVATE PN_AFC_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
