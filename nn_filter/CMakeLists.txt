cmake_minimum_required(VERSION 3.20)
project(m1pn_nn_filter LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(PARENT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
add_subdirectory("${PARENT_ROOT}" "${CMAKE_BINARY_DIR}/afc_core" EXCLUDE_FROM_ALL)
add_library(nn_filter INTERFACE)
target_include_directories(nn_filter INTERFACE src "${PARENT_ROOT}/nx/src")
target_link_libraries(nn_filter INTERFACE m1pn_core)
add_executable(m1pn_nn_filter src/main.cpp)
target_link_libraries(m1pn_nn_filter PRIVATE nn_filter)
target_compile_definitions(m1pn_nn_filter PRIVATE PN_AFC_ROOT="${PARENT_ROOT}" NN_FILTER_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(m1pn_unfiltered "${PARENT_ROOT}/nx/src/study_main.cpp")
target_include_directories(m1pn_unfiltered PRIVATE "${PARENT_ROOT}/nx/src")
target_link_libraries(m1pn_unfiltered PRIVATE m1pn_core)
target_compile_definitions(m1pn_unfiltered PRIVATE PN_AFC_ROOT="${PARENT_ROOT}")
enable_testing()
add_executable(nn_filter_checks tests/checks.cpp)
target_link_libraries(nn_filter_checks PRIVATE nn_filter)
target_compile_definitions(nn_filter_checks PRIVATE NN_FILTER_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
add_test(NAME nn_filter_algebra COMMAND nn_filter_checks)
set_tests_properties(nn_filter_algebra PROPERTIES ENVIRONMENT "OMP_NUM_THREADS=1;OPENBLAS_NUM_THREADS=1" TIMEOUT 180)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME nn_filter_cli COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.py" $<TARGET_FILE:m1pn_nn_filter>)
set_tests_properties(nn_filter_cli PROPERTIES ENVIRONMENT "OMP_NUM_THREADS=1;OPENBLAS_NUM_THREADS=1" TIMEOUT 180)
execute_process(COMMAND ${Python3_EXECUTABLE} -c "import torch"
  RESULT_VARIABLE TORCH_IMPORT_STATUS OUTPUT_QUIET ERROR_QUIET)
if(TORCH_IMPORT_STATUS EQUAL 0)
  add_test(NAME nn_filter_preprocessing COMMAND ${Python3_EXECUTABLE}
    "${CMAKE_CURRENT_SOURCE_DIR}/tests/preprocessing.py" $<TARGET_FILE:nn_filter_checks>)
  set_tests_properties(nn_filter_preprocessing PROPERTIES
    ENVIRONMENT "OMP_NUM_THREADS=1;OPENBLAS_NUM_THREADS=1" TIMEOUT 180)
endif()
